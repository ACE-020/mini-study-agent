% Measure one semantic unit. Keep short units together; let oversized ones flow.
\NewDocumentEnvironment{noteunit}{+b}{%
  \par\begingroup
  \measuringunittrue
  \begin{lrbox}{\unitcontents}\begin{minipage}{\linewidth}%
    \setlength{\parindent}{0pt}\setlength{\parskip}{4pt}%
    \MathSpacing\ignorespaces#1%
  \end{minipage}\end{lrbox}%
  \measuringunitfalse
  \ifdim\dimexpr\ht\unitcontents+\dp\unitcontents\relax>0.8\textheight
    \Needspace{5\baselineskip}%
    #1\par
  \else
    \Needspace{\dimexpr\ht\unitcontents+\dp\unitcontents\relax}%
    \noindent\usebox{\unitcontents}\par
  \fi
  \vspace{2pt}\endgroup
}{}
